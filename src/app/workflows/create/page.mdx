# Create a workflow {{ tag: 'POST', label: '/v1/workflows' }}

<Row>
  <Col>

    This endpoint allows you to create a new workflow. A name and at least one step are required to create a workflow.

    ### Required attributes

    <Properties>
      <Property name="name" type="string">
        Human-readable name for the workflow.
      </Property>
      <Property name="steps" type="array">
        List of step definitions. Each step must include an `agent` and `task`.
      </Property>
    </Properties>

    ### Optional attributes

    <Properties>
      <Property name="description" type="string">
        Description of what the workflow does.
      </Property>
      <Property name="timeout" type="integer">
        Maximum execution time in seconds. Defaults to `600`.
      </Property>
      <Property name="max_retries" type="integer">
        Number of times to retry failed steps. Defaults to `2`.
      </Property>
      <Property name="on_failure" type="string">
        Failure strategy: `"abort"` or `"continue"`.
      </Property>
    </Properties>

  </Col>
  <Col sticky>

    <CodeGroup title="Request" tag="POST" label="/v1/workflows">

    ```bash {{ title: 'cURL' }}
    curl -X POST http://localhost:8000/v1/workflows \
      -H "Authorization: Bearer {token}" \
      -H "Content-Type: application/json" \
      -d '{
        "name": "content-pipeline",
        "description": "Research, write, and review content.",
        "steps": [
          {"agent": "researcher", "task": "Research the topic"},
          {"agent": "writer", "task": "Write a draft", "depends_on": "researcher"},
          {"agent": "reviewer", "task": "Review the draft", "depends_on": "writer"}
        ]
      }'
    ```

    ```python {{ title: 'Python' }}
    from agentflow import AgentFlowClient

    client = AgentFlowClient(token)

    workflow = client.workflows.create(
        name="content-pipeline",
        description="Research, write, and review content.",
        steps=[
            {"agent": "researcher", "task": "Research the topic"},
            {"agent": "writer", "task": "Write a draft", "depends_on": "researcher"},
            {"agent": "reviewer", "task": "Review the draft", "depends_on": "writer"},
        ],
    )
    ```

    </CodeGroup>
